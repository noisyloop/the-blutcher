cmake_minimum_required(VERSION 3.22)
project(TheBlucher VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(BLUCHER_ENGINE_ONLY "Build dependency-free engine tests" OFF)
include(CTest)
add_executable(BlucherTests tests/EngineTests.cpp)
target_include_directories(BlucherTests PRIVATE Source)
add_test(NAME BlucherEngine COMMAND BlucherTests)
if(BLUCHER_ENGINE_ONLY)
  return()
endif()
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(TheBlucher COMPANY_NAME "Noisyloop" BUNDLE_ID "audio.noisyloop.blucher"
  PLUGIN_MANUFACTURER_CODE NsLp PLUGIN_CODE Blch FORMATS AU VST3 Standalone
  PRODUCT_NAME "The Blucher" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(TheBlucher)
target_sources(TheBlucher PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(TheBlucher PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TheBlucher PRIVATE juce::juce_audio_utils juce::juce_audio_formats juce::juce_gui_extra juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
